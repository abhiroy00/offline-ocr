# GPU OCR engine for the EC2 deployment box only.
#
# DO NOT install this blindly. Before installing, verify the environment on
# the target machine:
#
#   nvidia-smi                          # confirm driver + GPU are visible
#   python3 --version                   # PaddlePaddle 3.0 requires Python 3.9-3.12
#   nvcc --version  (or cat /usr/local/cuda/version.json)
#
# PaddlePaddle publishes prebuilt GPU wheels for specific CUDA versions only
# (CUDA 11.8 and CUDA 12.3 as of paddlepaddle-gpu 3.0.0). Pick the extra index
# URL that matches the CUDA version actually installed on the box -- do not
# guess. See https://www.paddlepaddle.org.cn/en/install/quick for the current
# matrix before changing this file.
#
# Example for CUDA 12.3:
#   pip install -r requirements/base.txt
#   pip install paddlepaddle-gpu==3.0.0 -i https://www.paddlepaddle.org.cn/packages/stable/cu123/
#   pip install paddleocr==2.9.1 shapely==2.0.6 pyclipper==1.3.0.post6
#
# Example for CUDA 11.8:
#   pip install paddlepaddle-gpu==3.0.0 -i https://www.paddlepaddle.org.cn/packages/stable/cu118/
#
# This file intentionally does NOT pin a CUDA-specific index so a blind
# `pip install -r requirements/ocr-gpu.txt` cannot install a mismatched wheel.
# Resolve the two commands above manually (or via deploy/install_gpu_ocr.sh,
# which detects CUDA and runs the right command for you).
setuptools>=68  # paddle imports this at runtime; not guaranteed present in a bare venv
paddleocr==2.9.1
shapely==2.0.6
pyclipper==1.3.0.post6
